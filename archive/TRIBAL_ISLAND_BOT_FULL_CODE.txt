import subprocess
import time
import os
import cv2
import numpy as np
import threading
import math


class ADBController:
    """
    ADBController quản lý giao tiếp ngầm 100% với BlueStacks:
    - Chụp màn hình trực tiếp từ bộ đệm Android (không dính cửa sổ Windows hay GUI)
    - Tự động tap (chạm) các nút Bắt Đầu, Tấn Công, Kỹ Năng, Hồi Máu, Chọn Perk
    - Vuốt Joystick di chuyển không chiếm chuột hay bàn phím máy tính
    """

    def __init__(self, adb_path=r"C:\Program Files\BlueStacks_nxt\HD-Adb.exe", device_id="emulator-5554"):
        self.adb_path = adb_path
        self.device_id = device_id
        
        # Tọa độ chuẩn 100% chuẩn xác trên màn hình 1920x1080
        self.COORD_ATTACK = (1757, 929)     # Nút Lửa to góc phải dưới
        self.COORD_SKILL = (1761, 643)      # Nút Kỹ năng tím phía trên nút lửa
        self.COORD_HEAL = (1445, 956)       # Nút Đùi gà hồi máu
        self.COORD_START = (980, 950)       # Nút Vàng Bắt Đầu / Tiếp Tục ở sảnh chờ
        self.COORD_END_BATTLE = (1442, 1000) # Nút Vàng 'Kết Thúc' sau khi hết trận
        
        # 3 thẻ chọn kỹ năng khi lên cấp (đo chính xác theo viền tròn trắng)
        self.PERK_COORDS = [
            (755, 701),   # Thẻ 1 (Trái)
            (943, 701),   # Thẻ 2 (Giữa)
            (1132, 701)   # Thẻ 3 (Phải)
        ]

        # Tọa độ Cần gạt di chuyển (Dynamic Joystick)
        self.JOYSTICK_CENTER = (350, 750)
        self.JOYSTICK_OFFSET = 160

    def run_cmd(self, args, timeout=15):
        """Chạy lệnh ADB với thiết bị chỉ định."""
        cmd = [self.adb_path, "-s", self.device_id] + args
        try:
            return subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, timeout=timeout)
        except subprocess.TimeoutExpired:
            print(f"[ADB] Lệnh bị timeout sau {timeout}s: {' '.join(args)}")
            return None
        except Exception as e:
            print(f"[ADB] Lỗi chạy lệnh: {e}")
            return None

    def ensure_connection(self):
        """Khởi tạo ADB daemon và kiểm tra kết nối."""
        # Bước 1: Khởi động ADB daemon
        try:
            subprocess.run([self.adb_path, "start-server"],
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=15)
        except Exception:
            pass
        
        # Bước 2: Kiểm tra thiết bị
        res = self.run_cmd(["shell", "echo", "ok"])
        if res and res.returncode == 0 and "ok" in res.stdout:
            return True
        return False

    def tap(self, x, y):
        """Chạm cảm ứng vào tọa độ (x, y) ngầm."""
        self.run_cmd(["shell", "input", "tap", str(int(x)), str(int(y))])

    def swipe(self, x1, y1, x2, y2, duration_ms=1000):
        """Vuốt cảm ứng từ (x1, y1) tới (x2, y2) trong duration_ms."""
        self.run_cmd(["shell", "input", "swipe", str(int(x1)), str(int(y1)), str(int(x2)), str(int(y2)), str(int(duration_ms))])

    def capture_screen(self):
        """
        Chụp ảnh trực tiếp từ Android framebuffer.
        Trả về ảnh numpy BGR (1920x1080) sắc nét, không bị dính bất kỳ cửa sổ nào của Windows.
        """
        temp_remote = "/sdcard/s.png"
        temp_local = os.path.join(os.path.dirname(os.path.dirname(__file__)), "assets", "adb_live_frame.png")
        
        # 1. Chụp trong android
        self.run_cmd(["shell", "screencap", "-p", temp_remote])
        # 2. Kéo về máy tính
        self.run_cmd(["pull", temp_remote, temp_local])
        
        if os.path.exists(temp_local):
            frame = cv2.imread(temp_local)
            return frame
        return None

    def attack(self):
        """Bắn đạn thường (chạm nút lửa)."""
        self.tap(*self.COORD_ATTACK)

    def use_skill(self):
        """Bấm chiêu đặc biệt."""
        self.tap(*self.COORD_SKILL)

    def heal(self):
        """Bơm máu đùi gà."""
        self.tap(*self.COORD_HEAL)

    def click_start(self):
        """Bấm nút Bắt Đầu / Tiếp Tục ở sảnh chờ."""
        self.tap(*self.COORD_START)

    def click_end_battle(self):
        """Bấm nút Kết Thúc màu vàng để nhận thưởng và về sảnh."""
        self.tap(*self.COORD_END_BATTLE)

    def select_perk(self, index=1):
        """Chọn perk lên cấp (index: 1, 2, 3)."""
        if 1 <= index <= 3:
            x, y = self.PERK_COORDS[index - 1]
            self.tap(x, y)

    def move_angle(self, angle_deg, distance_ratio=1.0, duration_ms=1000):
        """
        Điều khiển Joystick theo góc 360 độ tự do (Analog Vector):
        - angle_deg: 0=Phải, 90=Lên, 180=Trái, 270=Xuống (hoặc bất kỳ góc chéo nào: 45, 135, 225, 315,...)
        - distance_ratio: Độ lệch cần gạt (0.2 -> 1.0)
        - duration_ms: Thời gian giữ cần gạt di chuyển (milliseconds)
        """
        rad = math.radians(angle_deg)
        cx, cy = self.JOYSTICK_CENTER
        max_offset = self.JOYSTICK_OFFSET * min(1.0, max(0.2, distance_ratio))
        
        # Hệ tọa độ màn hình Android: Y hướng xuống dưới nên -sin(rad)
        tx = int(cx + max_offset * math.cos(rad))
        ty = int(cy - max_offset * math.sin(rad))
        self.swipe(cx, cy, tx, ty, duration_ms=duration_ms)

    def move_direction(self, direction='w', duration_ms=1500):
        """
        Điều khiển Joystick theo phím hướng hoặc tổ hợp chéo:
        'w': Lên, 'd': Phải, 's': Xuống, 'a': Trái
        'wd': Chéo trên phải, 'wa': Chéo trên trái, 'sd': Chéo dưới phải, 'sa': Chéo dưới trái
        """
        angle_map = {
            'w': 90.0,
            'd': 0.0,
            's': 270.0,
            'a': 180.0,
            'wd': 45.0,
            'dw': 45.0,
            'wa': 135.0,
            'aw': 135.0,
            'sd': 315.0,
            'ds': 315.0,
            'sa': 225.0,
            'as': 225.0
        }
        angle = angle_map.get(direction.lower(), 90.0)
        self.move_angle(angle, distance_ratio=1.0, duration_ms=duration_ms)

import os
import re
import json
import numpy as np
from PIL import Image, ImageGrab

try:
    import cv2
except ImportError:
    cv2 = None

try:
    import pytesseract
except ImportError:
    pytesseract = None


class VisionEngine:
    """
    VisionEngine chịu trách nhiệm:
    - Chụp màn hình BlueStacks / toàn màn hình
    - Đọc % thanh máu (HP bar) bằng kỹ thuật dải màu HSV và đo độ dài thanh đỏ
    - Nhận diện màn hình Lên Cấp (Level Up) & 3 kỹ năng lựa chọn qua Template Matching
    - Nhận diện số Tầng hiện tại qua OCR (pytesseract)
    - Nhận diện màn hình Kết thúc trận / Nhận rương
    - Nhận diện màn hình Sảnh Chờ (Lobby) & vị trí Nút Bắt Đầu / Tiếp Tục màu vàng
    """

    def __init__(self, config_path=None, assets_dir=None):
        base_dir = os.path.dirname(os.path.dirname(__file__))
        self.config_path = config_path or os.path.join(base_dir, "config", "settings.json")
        self.assets_dir = assets_dir or os.path.join(base_dir, "assets")
        self.settings = self._load_settings()

        # Cấu hình đường dẫn tesseract nếu có
        tesseract_exe = self.settings.get("ocr_tesseract_path", r"C:\Program Files\Tesseract-OCR\tesseract.exe")
        if pytesseract and os.path.exists(tesseract_exe):
            pytesseract.pytesseract.tesseract_cmd = tesseract_exe

        # Cache ảnh mẫu
        self.templates = {}
        self._load_templates()

        # Tọa độ chuẩn trên độ phân giải 1920x1080 (tự động co giãn theo tỉ lệ màn hình thực tế)
        # HP bar: x: 140 -> 350, y: 135 -> 170
        # Floor text: x: 275 -> 580, y: 60 -> 110
        self.ref_width = 1920
        self.ref_height = 1080

    def _load_settings(self):
        if os.path.exists(self.config_path):
            try:
                with open(self.config_path, "r", encoding="utf-8") as f:
                    return json.load(f)
            except Exception as e:
                print(f"[VisionEngine] Lỗi đọc settings: {e}")
        return {}

    def _load_templates(self):
        if not cv2 or not os.path.exists(self.assets_dir):
            return
        for file in os.listdir(self.assets_dir):
            if file.lower().endswith((".png", ".jpg", ".jpeg")):
                name = os.path.splitext(file)[0]
                img_path = os.path.join(self.assets_dir, file)
                img = cv2.imread(img_path, cv2.IMREAD_COLOR)
                if img is not None:
                    self.templates[name] = img
        print(f"[VisionEngine] Đã tải {len(self.templates)} ảnh mẫu từ thư mục assets.")

    def capture_screen(self, bbox=None):
        """
        Chụp màn hình (hoặc vùng bbox = (left, top, right, bottom)).
        Trả về ảnh numpy BGR (cho OpenCV) và PIL Image.
        """
        screenshot = ImageGrab.grab(bbox=bbox)
        frame_rgb = np.array(screenshot)
        if cv2 is not None:
            frame_bgr = cv2.cvtColor(frame_rgb, cv2.COLOR_RGB2BGR)
        else:
            frame_bgr = frame_rgb
        return frame_bgr, screenshot

    def match_template(self, image, template_name, threshold=0.75):
        """
        Tìm ảnh mẫu template_name trong image bằng cv2.matchTemplate.
        Trả về (found: bool, max_val: float, (max_x, max_y))
        """
        if cv2 is None or template_name not in self.templates:
            return False, 0.0, (0, 0)

        template = self.templates[template_name]
        
        # Co giãn template theo kích thước ảnh thực tế so với chuẩn 1920x1080
        h_img, w_img = image.shape[:2]
        if w_img != self.ref_width or h_img != self.ref_height:
            scale_x = w_img / self.ref_width
            scale_y = h_img / self.ref_height
            new_w = max(1, int(template.shape[1] * scale_x))
            new_h = max(1, int(template.shape[0] * scale_y))
            template_scaled = cv2.resize(template, (new_w, new_h))
        else:
            template_scaled = template

        if template_scaled.shape[0] > image.shape[0] or template_scaled.shape[1] > image.shape[1]:
            return False, 0.0, (0, 0)

        res = cv2.matchTemplate(image, template_scaled, cv2.TM_CCOEFF_NORMED)
        min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(res)

        if max_val >= threshold:
            return True, max_val, max_loc
        return False, max_val, max_loc

    def find_lobby_start_button(self, image):
        """
        Tìm nút vàng 'Bắt Đầu' hoặc 'Tiếp Tục' ở sảnh chờ để tự động vào trận.
        Hỗ trợ cả 2 trạng thái:
        - 'Bắt Đầu' (khi bắt đầu lượt mới từ tầng 1)
        - 'Tiếp Tục' (khi tiếp tục ván đấu dang dở)
        Trả về: (found: bool, (click_x, click_y))
        """
        if cv2 is None:
            return False, None

        h, w = image.shape[:2]

        # Lớp 1: Template Matching với cả 2 mẫu nút: sample_btn_batdau và sample_start_btn
        for tmpl_name in ["sample_btn_batdau", "sample_start_btn", "sample_btn_batdau_1080"]:
            if tmpl_name in self.templates:
                template = self.templates[tmpl_name]
                # Co dãn template về tỉ lệ kích thước ảnh chụp
                scale_x = w / 1024 if template.shape[1] < 300 else w / self.ref_width
                scale_y = h / 473 if template.shape[0] < 80 else h / self.ref_height
                t_w = max(1, int(template.shape[1] * scale_x))
                t_h = max(1, int(template.shape[0] * scale_y))
                if t_w < w and t_h < h:
                    t_scaled = cv2.resize(template, (t_w, t_h))
                    res = cv2.matchTemplate(image, t_scaled, cv2.TM_CCOEFF_NORMED)
                    _, max_val, _, max_loc = cv2.minMaxLoc(res)
                    if max_val >= 0.75:
                        click_x = max_loc[0] + t_w // 2
                        click_y = max_loc[1] + t_h // 2
                        return True, (click_x, click_y)

        # Lớp 2: Quét dải màu vàng nút bấm ở khu vực đáy màn hình (y: 75% -> 98%, x: 30% -> 70%)
        hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
        mask = cv2.inRange(hsv, (18, 120, 150), (35, 255, 255))
        roi_mask = np.zeros_like(mask)
        roi_mask[int(h * 0.75):int(h * 0.98), int(w * 0.30):int(w * 0.70)] = mask[int(h * 0.75):int(h * 0.98), int(w * 0.30):int(w * 0.70)]

        contours, _ = cv2.findContours(roi_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        for cnt in contours:
            x, y, bw, bh = cv2.boundingRect(cnt)
            # Nút bấm là hình chữ nhật rộng nằm ngang
            if bw > w * 0.15 and bh > h * 0.05:
                click_x = x + bw // 2
                click_y = y + bh // 2
                return True, (click_x, click_y)

        return False, None

    def get_hp_percentage(self, image):
        """
        Đo % máu thực tế bằng cách kiểm tra chiều dài phần màu hồng/đỏ của thanh HP.
        Chuẩn tọa độ trên 1920x1080: x: 140 -> 350, y: 135 -> 170.
        """
        if cv2 is None:
            return 100.0

        h, w = image.shape[:2]
        x1 = int(w * (140 / self.ref_width))
        x2 = int(w * (350 / self.ref_width))
        y1 = int(h * (135 / self.ref_height))
        y2 = int(h * (170 / self.ref_height))

        hp_crop = image[y1:y2, x1:x2]
        if hp_crop.size == 0:
            return 100.0

        # Lọc màu đỏ / hồng tím (dải máu) trong không gian HSV
        hsv = cv2.cvtColor(hp_crop, cv2.COLOR_BGR2HSV)
        mask1 = cv2.inRange(hsv, np.array([150, 80, 80]), np.array([180, 255, 255]))
        mask2 = cv2.inRange(hsv, np.array([0, 80, 80]), np.array([15, 255, 255]))
        mask = mask1 | mask2

        # Kiểm tra các cột có chứa điểm ảnh màu máu
        col_has_red = np.any(mask > 0, axis=0)
        filled_width = np.sum(col_has_red)
        total_width = hp_crop.shape[1]

        if total_width == 0:
            return 100.0

        percent = (filled_width / total_width) * 100.0
        return min(100.0, max(0.0, percent))

    def detect_floor_number(self, image):
        """
        Đọc số tầng hiện tại từ góc trên trái ("Đảo Nấm Tầng [X]").
        Chuẩn tọa độ 1920x1080: x: 275 -> 580, y: 60 -> 110.
        """
        h, w = image.shape[:2]
        x1 = int(w * (275 / self.ref_width))
        x2 = int(w * (580 / self.ref_width))
        y1 = int(h * (60 / self.ref_height))
        y2 = int(h * (110 / self.ref_height))

        crop = image[y1:y2, x1:x2]
        if crop.size == 0:
            return None

        # 1. Nếu có tesseract OCR
        if pytesseract is not None:
            gray = cv2.cvtColor(crop, cv2.COLOR_BGR2GRAY) if cv2 else crop
            if cv2:
                _, thresh = cv2.threshold(gray, 170, 255, cv2.THRESH_BINARY)
            else:
                thresh = gray
            try:
                text = pytesseract.image_to_string(thresh, config="--psm 6")
                match = re.search(r'(?:Tầng|Tang|Floor|F)?\s*[:\.]?\s*(\d+)', text, re.IGNORECASE)
                if match:
                    floor = int(match.group(1))
                    if 1 <= floor <= 150:
                        return floor
            except Exception:
                pass

        return None

    def is_level_up_screen(self, image):
        """
        Kiểm tra màn hình lên cấp (dải băng 'Lần này mạo hiểm tăng đến' hoặc 3 thẻ kỹ năng).
        Độ chính xác mẫu đạt 1.000 với threshold 0.70.
        """
        found, val, _ = self.match_template(image, "sample_level_up", threshold=0.70)
        if found:
            return True
            
        # Dự phòng: Kiểm tra sự xuất hiện của các viền tròn trắng của 3 thẻ chọn kỹ năng
        # Vùng y: 600..800, x: 650..1250
        if cv2 is not None and image is not None:
            h, w = image.shape[:2]
            roi = image[int(h * 0.55):int(h * 0.78), int(w * 0.35):int(w * 0.65)]
            gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY)
            white_pixels = np.sum(gray > 220)
            # Khi 3 thẻ kỹ năng hiện ra, tổng số điểm ảnh trắng của viền tròn > 2000
            if white_pixels > 1800:
                return True

        return False

    def is_end_battle_screen(self, image):
        """
        Kiểm tra màn hình kết thúc trận (Chiến đấu kết thúc / Rương thưởng).
        """
        found_end, _, _ = self.match_template(image, "sample_end_battle", threshold=0.70)
        if found_end:
            return True

        found_btn, _, _ = self.match_template(image, "sample_btn_ketthuc", threshold=0.70)
        if found_btn:
            return True

        found_chest, _, _ = self.match_template(image, "sample_chest", threshold=0.75)
        if found_chest:
            return True

        found_lobby_chest, _, _ = self.match_template(image, "sample_chest_lobby", threshold=0.75)
        if found_lobby_chest:
            return True

        return False

    def find_portal(self, image):
        """
        Tìm cổng dịch chuyển màu vàng ở tường trên khi dọn sạch quái.
        Trả về (found: bool, (portal_x, portal_y))
        """
        if cv2 is None or image is None:
            return False, None

        # Template matching với mẫu quả cầu dịch chuyển vàng
        found, val, loc = self.match_template(image, "sample_portal", threshold=0.70)
        if found:
            tpl = self.templates.get("sample_portal")
            tw = tpl.shape[1] if tpl is not None else 60
            th = tpl.shape[0] if tpl is not None else 60
            return True, (loc[0] + tw // 2, loc[1] + th // 2)

        # Lớp dự phòng: Lọc màu xoáy vàng phát sáng ở 40% trên cùng của màn hình
        h, w = image.shape[:2]
        roi = image[0:int(h * 0.40), int(w * 0.25):int(w * 0.75)]
        hsv = cv2.cvtColor(roi, cv2.COLOR_BGR2HSV)
        mask = cv2.inRange(hsv, (18, 140, 180), (35, 255, 255))
        cnts, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        for c in cnts:
            if 400 < cv2.contourArea(c) < 10000:
                M = cv2.moments(c)
                if M["m00"] > 0:
                    cx = int(M["m10"] / M["m00"]) + int(w * 0.25)
                    cy = int(M["m01"] / M["m00"])
                    return True, (cx, cy)

        return False, None

import time

import json
import os

try:
    import pydirectinput
    pydirectinput.PAUSE = 0.01

def tap_key(key, hold_time=0.06):
    """
    Giữ phím trong hold_time giây (mặc định 60ms) để giả lập Android/BlueStacks
    kịp ghi nhận tín hiệu chắc chắn 100%.
    """
    if not pydirectinput:
        return
    try:
        pydirectinput.keyDown(key)
        time.sleep(hold_time)
        pydirectinput.keyUp(key)
    except Exception as e:
        print(f"[CombatEngine] Lỗi bấm phím {key}: {e}")


class CombatEngine:
    """
    CombatEngine chịu trách nhiệm mô phỏng các thao tác bàn phím:
    - Di chuyển lên trên (W) khi mới vào tầng để thoát khỏi hốc cửa hẹp
    - CHIẾN THUẬT KITING (LÙA QUÁI VÒNG TRÒN): Chạy theo vòng W -> D -> S -> A (hoặc ngược lại)
      để quái dồn thành một cụm chạy sau lưng, né tối đa sát thương và chống kẹt góc.
    - Tự động spam nút đánh thường (J) với nhịp độ người thật (~4 lần/giây) tránh bị trượt đòn
    - Kích hoạt kỹ năng đặc biệt (Space)
    - Sử dụng đùi gà hồi máu khi nhận tín hiệu (L)
    - Chọn kỹ năng khi lên cấp (1, 2, 3)
    - Chế độ tự sát (đứng yên chờ quái hạ gục khi đã đạt tầng mục tiêu)
    """

    def __init__(self, key_config_path=None, settings_path=None):
        base_dir = os.path.dirname(os.path.dirname(__file__))
        self.key_config_path = key_config_path or os.path.join(base_dir, "config", "key_mapping.json")
        self.settings_path = settings_path or os.path.join(base_dir, "config", "settings.json")

        self.keys = self._load_keys()
        self.settings = self._load_settings()

        self.attack_interval = self.settings.get("attack_interval", 0.22)
        self.attack_hold_time = self.settings.get("attack_hold_time", 0.06)

        self.is_running = False
        self.is_paused = False
        self.suicide_mode = False  # Đứng yên để chết nhận rương
        
        self._thread = None
        self._lock = threading.Lock()
        self.current_direction = None

        # Chiến thuật Kiting (Vòng tròn xuôi/ngược chiều kim đồng hồ)
        self.kiting_patterns = [
            ['w', 'd', 's', 'a'],  # Xuôi chiều kim đồng hồ: Lên -> Phải -> Xuống -> Trái
            ['w', 'a', 's', 'd']   # Ngược chiều kim đồng hồ: Lên -> Trái -> Xuống -> Phải
        ]
        self.current_pattern_idx = 0
        self.kiting_step = 0

    def _load_keys(self):
        default_keys = {
            "movement": ["w", "a", "s", "d"],
            "attack": "j",
            "heal_item": "l",
            "special_skill": "space",
            "select_perk": ["1", "2", "3"]
        }
        if os.path.exists(self.key_config_path):
            try:
                with open(self.key_config_path, "r", encoding="utf-8") as f:
                    return {**default_keys, **json.load(f)}
            except Exception as e:
                print(f"[CombatEngine] Lỗi đọc key mapping: {e}")
        return default_keys

    def _load_settings(self):
        if os.path.exists(self.settings_path):
            try:
                with open(self.settings_path, "r", encoding="utf-8") as f:
                    return json.load(f)
            except Exception as e:
                print(f"[CombatEngine] Lỗi đọc settings: {e}")
        return {}

    def enter_room(self, duration=1.5):
        """
        Khi mới vào tầng hoặc bắt đầu lượt, nhân vật nằm trong hốc cửa hẹp.
        Cần giữ phím 'W' để chạy thẳng lên trên vào giữa phòng rộng.
        """
        if not pydirectinput or self.suicide_mode or self.is_paused:
            return
        print("[CombatEngine] 🚪 Xuất phát đầu tầng: Giữ [W] tiến vào giữa phòng...")
        with self._lock:
            self._release_all_movement_keys()
            pydirectinput.keyDown("w")
            self.current_direction = "w"

        attack_key = self.keys.get("attack", "j")
        start = time.time()
        while time.time() - start < duration:
            tap_key(attack_key, hold_time=self.attack_hold_time)
            time.sleep(self.attack_interval)

        with self._lock:
            pydirectinput.keyUp("w")
            self.current_direction = None
        print("[CombatEngine] ✅ Đã vào phòng rộng thành công!")

    def start(self, enter_first=True):
        if self._thread and self._thread.is_alive():
            return
        self.is_running = True
        self.is_paused = False
        self.suicide_mode = False
        self._thread = threading.Thread(target=self._combat_loop, args=(enter_first,), daemon=True)
        self._thread.start()
        print("[CombatEngine] Đã khởi động luồng chiến đấu Kiting (Lùa quái vòng tròn).")

    def stop(self):
        self.is_running = False
        self._release_all_movement_keys()
        print("[CombatEngine] Đã dừng luồng chiến đấu.")

    def pause(self):
        """Tạm dừng di chuyển và tấn công (dùng khi lên cấp hoặc load màn)."""
        self.is_paused = True
        self._release_all_movement_keys()
        print("[CombatEngine] Đã tạm dừng thao tác.")

    def resume(self):
        self.is_paused = False
        print("[CombatEngine] Tiếp tục thao tác chiến đấu.")

    def set_suicide_mode(self, enabled=True):
        self.suicide_mode = enabled
        if enabled:
            self._release_all_movement_keys()
            print("[CombatEngine] Bật chế độ TỰ SÁT: Đứng yên chờ kết thúc trận.")

    def trigger_heal(self):
        heal_key = self.keys.get("heal_item", "l")
        tap_key(heal_key, hold_time=0.06)
        print(f"[CombatEngine] -> 🍗 ĐÃ DÙNG ĐÙI GÀ HỒI MÁU (Phím {heal_key})")

    def trigger_skill(self):
        skill_key = self.keys.get("special_skill", "space")
        tap_key(skill_key, hold_time=0.06)
        print(f"[CombatEngine] -> ⚡ ĐÃ DÙNG KỸ NĂNG (Phím {skill_key})")

    def select_perk(self, index: int):
        perk_keys = self.keys.get("select_perk", ["1", "2", "3"])
        if 1 <= index <= len(perk_keys):
            key = perk_keys[index - 1]
            tap_key(key, hold_time=0.06)
            print(f"[CombatEngine] -> 🌟 ĐÃ CHỌN KỸ NĂNG {index} (Phím {key})")

    def _release_all_movement_keys(self):
        if not pydirectinput:
            return
        for k in self.keys.get("movement", ["w", "a", "s", "d"]):
            try:
                pydirectinput.keyUp(k)
            except Exception:
                pass
        self.current_direction = None

    def _kiting_next_direction(self):
        """
        Di chuyển theo chu kỳ vòng tròn để gom quái đi theo sau lưng (Kiting).
        Thỉnh thoảng đảo chiều quay (xuôi / ngược kim đồng hồ) để tránh kẹt địa hình.
        """
        if not pydirectinput or self.suicide_mode or self.is_paused:
            return

        pattern = self.kiting_patterns[self.current_pattern_idx]
        next_direction = pattern[self.kiting_step % len(pattern)]
        self.kiting_step += 1

        # Cứ sau 1 vòng (4 bước), có 20% khả năng đổi chiều quay (clockwise <-> counter-clockwise)
        if self.kiting_step % len(pattern) == 0 and random.random() < 0.25:
            self.current_pattern_idx = 1 - self.current_pattern_idx
            # print("[CombatEngine] 🔄 Đảo chiều quay vòng tròn lùa quái!")

        with self._lock:
            if self.current_direction and self.current_direction != next_direction:
                try:
                    pydirectinput.keyUp(self.current_direction)
                except Exception:
                    pass
            try:
                pydirectinput.keyDown(next_direction)
            except Exception:
                pass
            self.current_direction = next_direction

    def _combat_loop(self, enter_first=True):
        if not pydirectinput:
            print("[CombatEngine] LỖI: Chưa cài pydirectinput!")
            return

        # Bước 1: Thoát khỏi hốc cửa hẹp nếu xuất phát
        if enter_first:
            self.enter_room(duration=1.5)

        attack_key = self.keys.get("attack", "j")
        last_dir_change = time.time()
        last_skill_trigger = time.time()
        
        self._kiting_next_direction()

        while self.is_running:
            if self.is_paused:
                time.sleep(0.1)
                continue

            if self.suicide_mode:
                time.sleep(0.2)
                continue

            now = time.time()

            # 1. Spam phím tấn công thường (J) với nhịp chuẩn người bấm (~4 lần/giây)
            tap_key(attack_key, hold_time=self.attack_hold_time)

            # 2. Dùng skill nhân vật (mỗi 4s)
            if now - last_skill_trigger > 4.0:
                self.trigger_skill()
                last_skill_trigger = now

            # 3. Chạy theo cung vòng tròn kiting (sau mỗi 2.0 - 2.8s chuyển cạnh tiếp theo)
            if now - last_dir_change > random.uniform(2.0, 2.8):
                self._kiting_next_direction()
                last_dir_change = now

            time.sleep(self.attack_interval)

        self._release_all_movement_keys()

import time
import threading
import pyautogui

try:
    import pygetwindow as gw
except ImportError:
    gw = None

from .engine_combat import CombatEngine
from .engine_vision import VisionEngine


class RunState:
    IDLE = "IDLE"
    LOBBY = "LOBBY"
    FIGHTING = "FIGHTING"
    LEVEL_UP = "LEVEL_UP"
    SUICIDE = "SUICIDE"
    BATTLE_END = "BATTLE_END"


def get_bluestacks_window():
    """
    Tìm chính xác cửa sổ game BlueStacks App Player,
    loại bỏ cửa sổ vô hình 'BlueStacks Keymap Overlay'.
    """
    if not gw:
        return None
    try:
        # 1. Ưu tiên tìm cửa sổ chứa App Player
        for w in gw.getAllWindows():
            title = w.title.strip()
            if "bluestacks" in title.lower() and "overlay" not in title.lower() and title != "":
                if "app player" in title.lower():
                    return w

        # 2. Dự phòng: bất kỳ cửa sổ bluestacks nào không chứa chữ 'overlay'
        for w in gw.getAllWindows():
            title = w.title.strip()
            if "bluestacks" in title.lower() and "overlay" not in title.lower() and title != "":
                return w
    except Exception:
        pass
    return None


class RunManager:
    """
    RunManager là bộ não điều phối:
    - Bắt đúng cửa sổ 'BlueStacks App Player' (không bắt nhầm 'Keymap Overlay')
    - Chụp ảnh đúng vùng cửa sổ BlueStacks (dù kéo cửa sổ sang góc nào cũng chính xác 100%)
    - Tự động click nút vàng [Bắt Đầu / Tiếp Tục]
    - Điều phối chu kỳ cày rương tự động hoàn toàn
    """

    def __init__(self, target_floor=71, hp_heal_threshold=30, auto_repeat=True, log_callback=None):
        self.target_floor = target_floor
        self.hp_heal_threshold = hp_heal_threshold
        self.auto_repeat = auto_repeat
        self.log_callback = log_callback or print

        self.combat = CombatEngine()
        self.vision = VisionEngine()

        self.state = RunState.IDLE
        self.current_floor = 1
        self.is_running = False
        self._thread = None

        self.total_runs_completed = 0
        self.bs_win = None

    def log(self, message):
        if self.log_callback:
            self.log_callback(message)

    def focus_bluestacks_window(self):
        """Kích hoạt và nổi cửa sổ BlueStacks App Player lên phía trước."""
        win = get_bluestacks_window()
        if win:
            self.bs_win = win
            try:
                if win.isMinimized:
                    win.restore()
                win.activate()
                # Click chuột vào thanh tiêu đề BlueStacks để ép Windows chuyển quyền focus 100%
                click_x = win.left + 200
                click_y = win.top + 15
                pyautogui.click(click_x, click_y)
                self.log(f"[RunManager] 🎯 Đã kích hoạt và chuyển quyền điều khiển vào: '{win.title}'")
                time.sleep(0.4)
                return True
            except Exception as e:
                self.log(f"[CẢNH BÁO] Không thể kích hoạt cửa sổ: {e}")
        else:
            self.log("[CẢNH BÁO] Không tìm thấy cửa sổ 'BlueStacks App Player'. Vui lòng mở game trước!")
        return False

    def get_bluestacks_bbox(self):
        """Lấy tọa độ vùng cửa sổ BlueStacks (left, top, right, bottom)."""
        win = get_bluestacks_window()
        if win and win.width > 200 and win.height > 200:
            self.bs_win = win
            return (win.left, win.top, win.right, win.bottom)
        return None

    def start(self):
        if self.is_running:
            return

        # 1. Kích hoạt đúng cửa sổ BlueStacks
        self.focus_bluestacks_window()

        self.is_running = True
        self.state = RunState.IDLE
        self.current_floor = 1
        self._thread = threading.Thread(target=self._run_loop, daemon=True)
        self._thread.start()
        self.log(f"[RunManager] Bắt đầu hệ thống Auto Bot. Tầng mục tiêu: {self.target_floor}")

    def stop(self):
        self.is_running = False
        self.state = RunState.IDLE
        self.combat.stop()
        self.log("[RunManager] Đã dừng toàn bộ bot.")

    def pause(self):
        self.combat.pause()
        self.log("[RunManager] Tạm dừng.")

    def resume(self):
        self.combat.resume()
        self.log("[RunManager] Tiếp tục.")

    def _click_start_lobby(self, click_coords, bbox=None):
        """Click vào nút màu vàng Tiếp Tục / Bắt Đầu ở Sảnh Chờ."""
        rx, ry = click_coords
        # Nếu đang chụp theo vùng cửa sổ, chuyển đổi tọa độ tương đối sang màn hình thực tế
        if bbox:
            screen_x = bbox[0] + rx
            screen_y = bbox[1] + ry
        else:
            screen_x, screen_y = rx, ry

        self.log(f"[RunManager] 🎯 Phát hiện nút [Bắt Đầu / Tiếp Tục] tại ({screen_x}, {screen_y}). Đang click...")
        try:
            # Đảm bảo cửa sổ được focus trước khi click
            if self.bs_win:
                try:
                    self.bs_win.activate()
                except Exception:
                    pass
            pyautogui.click(screen_x, screen_y)
        except Exception as e:
            self.log(f"[RunManager] Lỗi click chuột: {e}")

        self.log("[RunManager] Đang tải vào màn chơi (chờ 3.5 giây)...")
        time.sleep(3.5)

        # Chuyển trạng thái sang chiến đấu và vượt hốc cửa hẹp
        self.current_floor = 1
        self.state = RunState.FIGHTING
        self.combat.set_suicide_mode(False)
        self.combat.start(enter_first=True)

    def _run_loop(self):
        last_heal_time = 0
        last_floor_check_time = 0
        last_lobby_check_time = 0

        while self.is_running:
            now = time.time()
            
            # Lấy vùng cửa sổ BlueStacks thực tế trên màn hình
            bbox = self.get_bluestacks_bbox()
            frame_bgr, _ = self.vision.capture_screen(bbox=bbox)

            # 1. KIỂM TRA MÀN HÌNH SẢNH CHỜ (LOBBY) - TỰ ĐỘNG BẤM [BẮT ĐẦU / TIẾP TỤC]
            if self.state in [RunState.IDLE, RunState.LOBBY, RunState.BATTLE_END]:
                if now - last_lobby_check_time > 1.0:
                    found_btn, click_coords = self.vision.find_lobby_start_button(frame_bgr)
                    if found_btn:
                        self.state = RunState.LOBBY
                        self.combat.pause()
                        self._click_start_lobby(click_coords, bbox=bbox)
                        last_lobby_check_time = now
                        continue
                    last_lobby_check_time = now

            # 2. KIỂM TRA MÀN HÌNH KẾT THÚC TRẬN / NHẬN RƯƠNG
            if self.vision.is_end_battle_screen(frame_bgr):
                self.state = RunState.BATTLE_END
                self.combat.pause()
                self.log("[RunManager] 🎁 Phát hiện màn hình KẾT THÚC TRẬN ĐẤU / NHẬN RƯƠNG.")
                
                time.sleep(2.0)
                # Bấm xác nhận nhận quà (phím Space hoặc click)
                self.combat.trigger_skill()
                self.total_runs_completed += 1
                self.log(f"[RunManager] ✅ Hoàn thành lượt thứ {self.total_runs_completed}!")

                if self.auto_repeat:
                    self.log("[RunManager] Đang chờ hệ thống quay trở lại Sảnh Chờ để nhận diện nút Bắt Đầu...")
                    time.sleep(3.0)
                else:
                    self.stop()
                    break
                continue

            # 3. KIỂM TRA MÀN HÌNH LÊN CẤP (CHỌN KỸ NĂNG)
            if self.vision.is_level_up_screen(frame_bgr):
                if self.state != RunState.LEVEL_UP:
                    self.state = RunState.LEVEL_UP
                    self.combat.pause()
                    self.log("[RunManager] 🌟 LÊN CẤP! Tạm dừng đánh để chọn kỹ năng...")
                    
                    time.sleep(0.5)
                    self.combat.select_perk(1)
                    time.sleep(1.0)
                    
                    self.state = RunState.FIGHTING
                    self.combat.resume()
                    continue

            # 4. KIỂM TRA SỐ TẦNG ĐỊNH KỲ
            if self.state == RunState.FIGHTING and (now - last_floor_check_time > 2.0):
                detected_floor = self.vision.detect_floor_number(frame_bgr)
                if detected_floor is not None and detected_floor >= self.current_floor:
                    if detected_floor > self.current_floor:
                        self.current_floor = detected_floor
                        self.log(f"[RunManager] 🚪 >>> TIẾN VÀO TẦNG {self.current_floor} <<<")
                        # Thoát khỏi hốc cửa hẹp của tầng mới
                        self.combat.enter_room(duration=1.2)

                    # Kiểm tra xem đã đạt tầng mục tiêu chưa
                    if self.current_floor >= self.target_floor:
                        if not self.combat.suicide_mode:
                            self.state = RunState.SUICIDE
                            self.combat.set_suicide_mode(True)
                            self.log(f"[RunManager] 💀 !!! ĐÃ ĐẠT TẦNG MỤC TIÊU ({self.target_floor}) !!!")
                            self.log("[RunManager] Chuyển sang chế độ TỰ SÁT: Đứng yên chờ quái hạ gục để nhận rương.")
                last_floor_check_time = now

            # 5. KIỂM TRA VÀ BƠM MÁU
            if self.state == RunState.FIGHTING and not self.combat.suicide_mode:
                hp = self.vision.get_hp_percentage(frame_bgr)
                # Chỉ cảnh báo máu yếu khi hp > 0 (tránh trường hợp chưa đọc được màn hình game)
                if 0.0 < hp < self.hp_heal_threshold and (now - last_heal_time > 3.0):
                    self.log(f"[RunManager] 🍗 CẢNH BÁO MÁU YẾU ({hp:.1f}% < {self.hp_heal_threshold}%) -> BƠM ĐÙI GÀ!")
                    self.combat.trigger_heal()
                    last_heal_time = now

            # Nếu đang ở IDLE quá 3 giây mà không phải ở sảnh (người dùng bấm Start khi đã ở trong trận)
            if self.state == RunState.IDLE and self.is_running:
                # Kiểm tra xem có thanh máu không
                hp_test = self.vision.get_hp_percentage(frame_bgr)
                if hp_test > 10.0:
                    self.log("[RunManager] Phát hiện đang ở trong trận đấu! Chuyển sang chiến đấu...")
                    self.state = RunState.FIGHTING
                    self.combat.start(enter_first=False)

            time.sleep(0.1)

import cv2
import numpy as np
from typing import List, Tuple, Optional


class LineOfSightChecker:
    """
    LineOfSightChecker kiểm tra xem đường đạn giữa Người chơi và Quái vật
    có bị chặn bởi các vật cản (như Cọc gỗ, Tảng đá, Bờ tường) hay không.
    """

    def __init__(self):
        pass

    @staticmethod
    def is_segment_intersecting_circle(p1: Tuple[int, int], p2: Tuple[int, int],
                                       circle_center: Tuple[int, int], radius: int) -> bool:
        """
        Kiểm tra đoạn thẳng p1->p2 (đường đạn) có cắt qua hình tròn bán kính radius (vật cản cọc gỗ) không.
        """
        x1, y1 = p1
        x2, y2 = p2
        cx, cy = circle_center

        # Vector p1 -> p2
        dx = x2 - x1
        dy = y2 - y1
        line_len_sq = dx * dx + dy * dy

        if line_len_sq == 0:
            return (cx - x1)**2 + (cy - y1)**2 <= radius**2

        # Chiếu tâm hình tròn lên đoạn thẳng (từ 0.0 -> 1.0)
        t = max(0.0, min(1.0, ((cx - x1) * dx + (cy - y1) * dy) / line_len_sq))

        # Tọa độ điểm gần nhất trên đoạn thẳng
        closest_x = x1 + t * dx
        closest_y = y1 + t * dy

        dist_sq = (cx - closest_x)**2 + (cy - closest_y)**2
        return dist_sq <= (radius**2)

    def check_los(self, player_pos: Tuple[int, int], target_pos: Tuple[int, int],
                  obstacles: List[Tuple[int, int]], obstacle_radius: int = 45) -> Tuple[bool, Optional[Tuple[int, int]]]:
        """
        Kiểm tra tầm bắn (Line of Sight):
        - player_pos: Tọa độ người chơi (x, y)
        - target_pos: Tọa độ quái vật mục tiêu (x, y)
        - obstacles: Danh sách tọa độ tâm các cọc gỗ [(x, y), ...]
        - obstacle_radius: Bán kính an toàn của cọc gỗ

        Trả về:
        - has_clear_shot: True nếu đường bắn thông thoáng, False nếu bị chặn
        - blocking_obstacle: Tọa độ cọc gỗ gây cản trở (nếu có)
        """
        for obs in obstacles:
            if self.is_segment_intersecting_circle(player_pos, target_pos, obs, obstacle_radius):
                return False, obs
        return True, None

import sys
import os
import argparse

# Đảm bảo đường dẫn thư mục gốc nằm trong sys.path
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
if BASE_DIR not in sys.path:
    sys.path.insert(0, BASE_DIR)

from core.manager_run import RunManager


def parse_arguments():
    parser = argparse.ArgumentParser(description="Tribal Island Auto Bot")
    parser.add_argument("--cli", action="store_true", help="Chạy chế độ dòng lệnh (không cần giao diện GUI)")
    parser.add_argument("--floor", type=int, default=71, help="Tầng mục tiêu kết thúc lượt (mặc định: 71)")
    parser.add_argument("--hp", type=int, default=30, help="Phần trăm máu để kích hoạt bơm đùi gà (mặc định: 30)")
    parser.add_argument("--repeat", action="store_true", default=True, help="Tự động lặp lại sau khi kết thúc lượt")
    return parser.parse_args()


def run_cli_mode(manager):
    print("=" * 60)
    print("KHỞI ĐỘNG TRIBAL ISLAND BOT Ở CHẾ ĐỘ DÒNG LỆNH (CLI)")
    print(f"Mục tiêu tầng: {manager.target_floor} | Ngưỡng máu: {manager.hp_heal_threshold}%")
    print("Nhấn Ctrl + C để dừng bot bất kỳ lúc nào.")
    print("=" * 60)
    try:
        manager.start()
        while True:
            import time
            time.sleep(1)
    except KeyboardInterrupt:
        print("\n[!] Đang dừng bot...")
        manager.stop()
        print("[!] Đã thoát an toàn.")


def main():
    # Tạo thư mục assets nếu chưa có
    assets_dir = os.path.join(BASE_DIR, "assets")
    os.makedirs(assets_dir, exist_ok=True)

    args = parse_arguments()
    manager = RunManager(
        target_floor=args.floor,
        hp_heal_threshold=args.hp,
        auto_repeat=args.repeat
    )

    if args.cli:
        run_cli_mode(manager)
    else:
        try:
            from gui.app_window import launch_gui, PYSIDE_AVAILABLE
            if PYSIDE_AVAILABLE:
                launch_gui(manager)
            else:
                print("[!] PySide6 chưa được cài đặt. Tự động chuyển sang chế độ CLI...")
                run_cli_mode(manager)
        except Exception as e:
            print(f"[!] Không thể mở giao diện GUI ({e}). Chuyển sang CLI...")
            run_cli_mode(manager)


if __name__ == "__main__":
    main()

## 1. TỔNG QUAN MỤC TIÊU & TRIẾT LÝ KIẾN TRÚC

### 1.1. Hiện trạng & Bài toán thực tế
- Bot hiện tại đã vượt qua rào cản kỹ thuật lớn nhất: **Điều khiển chạy ngầm 100% qua ADB** (không chiếm chuột/bàn phím), nhận diện các trạng thái sảnh, chọn kỹ năng đa lần, hồi máu và kiting cơ bản.
- **Vấn đề cốt tử mới phát sinh (Minh chứng từ ảnh thực tế):** Khi nhân vật di chuyển theo mẫu hình cứng (Fixed Pattern W-D-S-A), nhân vật dễ bị lọt vào góc hẹp bị che khuất bởi vật cản (như hàng cọc gỗ tròn, bờ tường băng). Do cơ chế tự ngắm (Auto-aim) của game, đạn bắn thẳng vào cọc gỗ thay vì trúng quái, và bot không biết tự thoát khỏi góc kẹt này để mở góc bắn.
- **Mục tiêu:** Bổ sung tầng **Nhận thức không gian (Perception)**, **Bản đồ hóa chi phí (Cost Grid Map)** và **Điều hướng thích ứng (Adaptive Navigation)** để bot tự biết quái ở đâu, vật cản ở đâu, và tự tìm góc bắn thông thoáng nhất.

### 1.2. Triết lý kiến trúc
1. **Preserve Working Core (Bảo toàn lõi):** Giữ nguyên toàn bộ mã nguồn ADB Controller, Vision Engine cơ bản, và GUI. Tuyệt đối không viết lại các tính năng đã chạy chuẩn.
2. **Loosely Coupled, Interface-driven:** Mỗi module mới chỉ làm một việc duy nhất và giao tiếp qua dataclass chuẩn.
3. **Graceful Fallback (Dự phòng an toàn):** Nếu tầng AI nhận diện gặp sự cố hoặc độ tin cậy < 40%, bot lập tức kích hoạt cơ chế lùi về Fixed Kiting để tiếp tục chiến đấu mà không bao giờ bị đứng im.

```
│       ├── stuck_detector.py   # Phát hiện kẹt góc / kẹt cọc gỗ
│       └── vector_steering.py  # Tính vector di chuyển 360° gửi vào ADB
```

### Chi tiết nhiệm vụ từng module:
1. **`detector_enemy.py`:**
   - Dựa trên dải màu đặc trưng của quái (thanh máu đỏ mini trên đầu quái, thân quái xanh/đỏ/xám) để trích xuất danh sách tọa độ quái $(x_i, y_i)$ trên màn hình 1920x1080.
2. **`detector_hazard.py`:**
   - Phân đoạn nhận diện cọc gỗ tròn (đầu cọc có vân gỗ tròn xoắn ốc) và bẫy gai trên sàn cỏ.
3. **`line_of_sight.py`:**
   - Dựng đoạn thẳng nối từ Người chơi $\rightarrow$ Quái mục tiêu gần nhất. Kiểm tra xem đoạn thẳng này có cắt qua bất kỳ cọc gỗ/vật cản nào không. Nếu có $\rightarrow$ Báo động **Blocked LOS (Mất tầm nhìn đạn)**.
4. **`stuck_detector.py`:**
   - Nếu trong 2.0 giây người chơi không thay đổi vị trí đáng kể hoặc bị Blocked LOS liên tục $\rightarrow$ Sinh vector giải cứu (Break-out Vector) giật ngược hướng để thoát ra bãi đất trống.
5. **`vector_steering.py`:**
   - Sử dụng giải thuật trường thế nhân tạo (Artificial Potential Field):
     - **Lực đẩy (Repulsion):** Bị đẩy ra xa khỏi quái quá gần, cọc gỗ, bờ tường.
     - **Lực hút (Attraction):** Bị hút về vùng cỏ trống và góc có đường đạn thông thoáng (Clear LOS).
     - Kết quả: Góc $\theta \in [0^\circ, 360^\circ)$ tối ưu được truyền thẳng vào `adb.move_angle()`.

## 5. ĐẶC TẢ INTERFACE & GIAO THỨC TRUYỀN DỮ LIỆU

Để đảm bảo tính độc lập và khả năng kiểm thử đơn vị (Unit Test), toàn bộ giao tiếp giữa các tầng sử dụng cấu trúc dữ liệu chuẩn (`dataclass`):

```python
from dataclasses import dataclass, field
from typing import List, Tuple, Optional

@dataclass
class Entity:
    """Thực thể trong game (Quái, Người chơi, Cọc gỗ)."""
    id: int
    category: str        # 'player', 'enemy', 'boss', 'obstacle_stump', 'obstacle_wall'
    center_x: int        # Tọa độ pixel trên màn 1920x1080
    center_y: int
    radius: int = 40
    hp_percent: float = 100.0

@dataclass
class PerceptionResult:
    """Kết quả phân tích thị giác toàn diện của 1 khung hình."""
    timestamp: float
    player_pos: Tuple[int, int]
    enemies: List[Entity] = field(default_factory=list)
    obstacles: List[Entity] = field(default_factory=list)
    has_clear_shot: bool = True           # Tầm đạn tới quái gần nhất có bị che không
    nearest_enemy_dist: float = 9999.0

@dataclass
class SteeringCommand:
    """Lệnh điều khiển được chuyển xuống ADB."""
    action: str          # 'KITE', 'BREAKOUT', 'ENTER_DOOR', 'HOLD'
    angle_degrees: float # 0.0 -> 360.0 độ
    duration_ms: int     # Thời gian vuốt cần gạt
    should_attack: bool = True
    should_skill: bool = False
```

## 7. TIÊU CHÍ NGHIỆM THU (ACCEPTANCE CRITERIA)

Mỗi giai đoạn chỉ được coi là hoàn thành khi thỏa mãn các bài test thực nghiệm trên BlueStacks:

1. **Nghiệm thu P1 (Line of Sight):**
   - Khi đặt nhân vật sau hàng cọc gỗ (như trong ảnh lỗi), hệ thống phải xuất ra log: `[LOS] CẢNH BÁO: Đạn bị cản bởi cọc gỗ tại (X, Y)`.
2. **Nghiệm thu P2 (Stuck Recovery):**
   - Trong vòng 1.5 giây sau khi bị kẹt, bot phải tự động vuốt cần gạt theo hướng thoáng (S hoặc A) để đưa nhân vật ra vùng cỏ rộng, không bao giờ đứng im bắn vào cọc gỗ.
3. **Nghiệm thu P5 (Continuous Combat):**
   - Vòng lặp chiến đấu không bị dừng sau 15 giây. Bot tiếp tục vừa chạy vừa bắn cho đến khi quái vật cuối cùng trên sàn ngã xuống.
4. **Nghiệm thu P7 (Floor Progression):**
   - Bot tự đi qua ít nhất 3 tầng liên tiếp (Tầng 1 $\rightarrow$ Tầng 2 $\rightarrow$ Tầng 3) mà không cần can thiệp tay của con người.
